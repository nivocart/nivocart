# NivoCart: block script execution inside the upload area.
# Uploaded files here must only ever be served as static files.
<FilesMatch "(?i)\.(php[0-9]*|phtml|phar|pht|phps|pl|py|cgi|sh|asp|aspx|jsp|shtml)$">
	<IfModule mod_authz_core.c>
		Require all denied
	</IfModule>
	<IfModule !mod_authz_core.c>
		Order allow,deny
		Deny from all
	</IfModule>
</FilesMatch>

<IfModule mod_php.c>
	php_flag engine off
</IfModule>
<IfModule mod_php7.c>
	php_flag engine off
</IfModule>
<IfModule mod_php8.c>
	php_flag engine off
</IfModule>
